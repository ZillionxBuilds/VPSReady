hostnamectl --static
timedatectl show --property=Timezone --value
findmnt --bytes -n -o SOURCE,SIZE,USED,AVAIL,USE%,TARGET --target /
sudo -n ufw status numbered
sudo -n dpkg --audit
# เฉพาะก่อน/หลัง reboot test เก็บค่าจริง local
